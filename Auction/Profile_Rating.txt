public static int UpdateProfileRating(int EntityId, HelperConstants.UserType objUserType)
        {
            int errcode = 0;
            IDBConnection lobjConn = DBManager.Open(DataAccessSettings.GetDataAccessSettings().DefaultDatabase, false, Provider.SQL);
            IDBCommand oCmd = lobjConn.CreateCommand(StoredProcedures.ProfileRatingReset, CommandType.StoredProcedure);
            try
            {
                oCmd[DBFields.EntityId] = EntityId;
                oCmd[DBFields.EntityType] = objUserType.ToString();
                errcode = Convert.ToInt32(oCmd.ExecuteNonQuery());
            }
            catch (Exception ex)
            {
                LoggingAdapter.WriteLog("Profile Ratings Reset" + ex.Message + Environment.NewLine);
            }
            finally
            {
                lobjConn.Close();
                lobjConn = null;
            }
            return errcode;
        }

    //[OperationBehavior(TransactionScopeRequired = true, TransactionAutoComplete = true)]
        public ServiceResponse InsertVendorAdditionDet(IList<VendorDetails> lobjVendorDetails)
        {
            IList<Customer> objCustomer = new List<Customer>();
            ServiceResponse response = new ServiceResponse();
            try
            {
                InfiAuctionFacade lobjInfiAuctionFacade = new InfiAuctionFacade();
                //for (int i = 0; i < lobjVendorDetails.Count;i++)
                foreach (VendorDetails objVendorDetails in lobjVendorDetails)
                {
                    response = lobjInfiAuctionFacade.InsertVendorAdditionDet(objVendorDetails);
                    if (response.Errcode <= 0)
                    {
                        return response;
                    }
                }
                objCustomer = lobjInfiAuctionFacade.GetCustomerById(lobjVendorDetails[0].CreatedBy);
                switch (objCustomer[0].EntityType)
                {
                    case HelperConstants.UserType.CMP:
                        lobjInfiAuctionFacade.UpdateProfileRating(lobjVendorDetails[0].EntityId, HelperConstants.UserType.CMP);
                        RefreshCache(HelperConstants.CacheKey.Company);
                        break;
                    case HelperConstants.UserType.VND:
                        lobjInfiAuctionFacade.UpdateProfileRating(lobjVendorDetails[0].EntityId, HelperConstants.UserType.VND);
                        RefreshCache(HelperConstants.CacheKey.Vendor);
                        break;
                }
                
            }
            catch (Exception ex)
            {
                LoggingAdapter.WriteLog("Insert Vendor Addition Details:" + ex.Message + Environment.NewLine + ex.StackTrace);
            }
            return response;
        }
        #endregion
